cmake_minimum_required(VERSION 3.22)
project(KMX6 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_subdirectory(third_party/JUCE)
juce_add_binary_data(KMX6Credits NAMESPACE CreditsAssets HEADER_NAME CreditsAssets.h SOURCES Assets/Credits/Juce.png Assets/Credits/VirtualCables.png Assets/KMX6Icon.png)
juce_add_gui_app(KMX6 ICON_BIG Assets/KMX6Icon.png ICON_SMALL Assets/KMX6Icon.png PRODUCT_NAME "KMX6" COMPANY_NAME "KMX6 Project" BUNDLE_ID "com.kmx6projects.mixer")
juce_generate_juce_header(KMX6)
target_link_libraries(KMX6 PRIVATE KMX6Credits)
target_sources(KMX6 PRIVATE Source/Main.cpp Source/AudioCore.h Source/NativeCapture.h Source/NativeCapture.cpp)
target_compile_definitions(KMX6 PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_VST=0 JUCE_ASIO=1 JUCE_WASAPI=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0 NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(KMX6 PRIVATE /utf-8)
endif()
target_link_libraries(KMX6 PRIVATE juce::juce_audio_utils juce::juce_dsp avrt ole32 uuid propsys shell32 gdi32 user32 dwmapi PUBLIC juce::juce_recommended_config_flags)

add_library(KMX6WASAPI STATIC third_party/miniaudio/miniaudio.c)
target_include_directories(KMX6WASAPI PUBLIC third_party/miniaudio)
target_compile_definitions(KMX6WASAPI PUBLIC MA_NO_DECODING MA_NO_ENCODING MA_NO_ENGINE MA_NO_RESOURCE_MANAGER MA_NO_NODE_GRAPH MA_ENABLE_ONLY_SPECIFIC_BACKENDS MA_ENABLE_WASAPI)
target_link_libraries(KMX6WASAPI PRIVATE ole32 uuid avrt winmm)
target_link_libraries(KMX6 PRIVATE KMX6WASAPI)
# Narrow setup helper: merge only KMX6's attachments, preserving other USB/IP devices.
add_executable(KMX6DeviceSetup Source/KMX6DeviceSetup.cpp)
target_include_directories(KMX6DeviceSetup PRIVATE tools/USBip/include)
target_link_libraries(KMX6DeviceSetup PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tools/USBip/lib/libusbip.lib")
target_compile_definitions(KMX6DeviceSetup PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_custom_command(TARGET KMX6DeviceSetup POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/tools/USBip/libusbip.dll" "$<TARGET_FILE_DIR:KMX6DeviceSetup>")
